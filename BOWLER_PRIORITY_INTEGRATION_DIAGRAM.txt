  /// 🎯 PRIORITY UPDATE: Fetch and update ONLY bowler name FIRST after over completion
  /// This ensures bowler name is displayed before other LED updates
  Future<void> _updateBowlerNamePriority() async {
    try {
      final bleService = BleManagerService();

      if (!bleService.isConnected) {
        debugPrint('⚠️ Bluetooth not connected. Skipping bowler name priority update.');
        return;
      }

      debugPrint('🎳 [PRIORITY] Fetching bowler name for immediate update...');

      // Get bowler name (prioritized)
      final bowlerPlayer = TeamMember.getByPlayerId(currentBowler!.playerId);

      String truncateName(String name) {
        final truncated = name.length > 6 ? name.substring(0, 6) : name;
        return truncated.toUpperCase();
      }

      final bowlerName = truncateName(bowlerPlayer?.teamName ?? 'BOWL1');

      // Fixed position for bowler name
      const int bowlerNameX = 10, bowlerNameY = 60, bowlerNameWidth = 32, bowlerNameHeight = 10;

      // ⚡ IMMEDIATE: Scroll out, update, scroll in bowler name ONLY
      debugPrint('📤 [PRIORITY] Scrolling out bowler name...');

      // Scroll out bowler name (left)
      List<String> scrollOut = [
        'SCROLL $bowlerNameX $bowlerNameY $bowlerNameWidth $bowlerNameHeight 150 left',
      ];
      await bleService.sendRawCommands(scrollOut);

      // Wait for scroll out to complete (150ms)
      await Future.delayed(const Duration(milliseconds: 150));

      // Update bowler name (instant)
      List<String> updateName = [
        'CHANGE $bowlerNameX $bowlerNameY $bowlerNameWidth $bowlerNameHeight 1 255 200 200 $bowlerName',
      ];
      await bleService.sendRawCommands(updateName);

      // Scroll in bowler name (right)
      List<String> scrollIn = [
        'SCROLL $bowlerNameX $bowlerNameY $bowlerNameWidth $bowlerNameHeight 150 right',
      ];
      await bleService.sendRawCommands(scrollIn);

      debugPrint('✅ [PRIORITY] Bowler name "$bowlerName" updated and visible first (300ms total)');

    } catch (e) {
      debugPrint('❌ Bowler name priority update failed: $e');
    }
  }
